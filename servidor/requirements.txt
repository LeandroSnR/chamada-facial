# Servidor — roda na máquina que processa o reconhecimento facial
# Instalação: pip install -r requirements.txt
#
# Nota: setuptools<81 é necessário porque o pacote face_recognition_models
# (dependência do face_recognition) usa a API legada `pkg_resources`,
# removida a partir do setuptools 81. Sem esse pin, o face_recognition
# falha ao importar com "Please install face_recognition_models...".
setuptools<81

fastapi==0.115.0
uvicorn[standard]==0.30.6

# Usamos a versão "headless" porque o servidor nunca abre janela gráfica
# (só processa frames recebidos via API). A versão não-headless exige a
# biblioteca de sistema libGL, que não existe em ambientes sem interface
# gráfica (containers, Codespaces, a maioria dos servidores Linux).
opencv-python-headless==4.10.0.84

face_recognition==1.3.0
numpy==1.26.4
python-multipart==0.0.9
python-dotenv==1.0.1
pydantic==2.9.2
